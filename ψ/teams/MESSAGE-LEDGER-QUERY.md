```python
# claim รูป "N ต่อ M" ที่บ้านหนึ่งส่งออกในเดือนนี้ — normalize identity ก่อนเสมอ (T1)
import sqlite3, re
con = sqlite3.connect('file:/home/user/.maw/message-ledger.sqlite?mode=ro', uri=True)
def stem(fid):                      # T1
    s = re.sub(r'^local:', '', fid or '')
    s = re.sub(r':[^:]*\.\d+$', '', s)          # cell:role.0
    s = re.sub(r'-(oracle|cell|complex-software-v\d+|dev-v\d+)$', '', s)
    return s.split('-')[0]
rows = con.execute("select ts, from_id, to_id, state, text from messages "
                   "where ts >= ? order by ts", ('2026-08-01',)).fetchall()   # T4: ts ไม่ใช่ id
NUM = re.compile(r'\b\d+\s*(?:จาก|of|/|ต่อ)\s*\d+\b')
for ts, f, t, st, txt in rows:
    if stem(f) == 'lucifer' and NUM.search(txt or ''):
        print(ts, '(UTC — +7 = local)', st, '->', t)                          # T2, T3
```
